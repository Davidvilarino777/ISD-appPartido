4. Recoger entradas:

-collect 1 '1111222233334444' // Entradas recogidas
mvn exec:java -Dexec.mainClass="es.udc.ws.app.client.ui.AppServiceClient" -Dexec.args="-collect 1 '1111222233334444'"

-collect 1 '1111222233334444' // Falla (entradas ya recogidas)
mvn exec:java -Dexec.mainClass="es.udc.ws.app.client.ui.AppServiceClient" -Dexec.args="-collect 1 '1111222233334444'"

-collect 2 '1111222233334444' // Falla (tarjeta incorrecta)
mvn exec:java -Dexec.mainClass="es.udc.ws.app.client.ui.AppServiceClient" -Dexec.args="-collect 2 '1111222233334444'"

-collect 9 '1111222233334444' // Falla (compra no existe)
mvn exec:java -Dexec.mainClass="es.udc.ws.app.client.ui.AppServiceClient" -Dexec.args="-collect 9 '1111222233334444'"

5. Buscar partidos por fecha:

-findMatches '2024-09-01' // Devuelve partido con id 1
mvn exec:java -Dexec.mainClass=es.udc.ws.app.client.ui.AppServiceClient "-Dexec.args=-findmatches '2024-09-01'T'00:00'"

-findMatches '2024-11-01' // Devuelve partidos con id 1 y 3
mvn exec:java -Dexec.mainClass=es.udc.ws.app.client.ui.AppServiceClient "-Dexec.args=-findmatches '2024-11-01'T'00:00'"

-findMatches '2024-08-01' // Devuelve lista vacía
mvn exec:java -Dexec.mainClass=es.udc.ws.app.client.ui.AppServiceClient "-Dexec.args=-findmatches '2024-08-01'T'00:00'"

-findMatches '2023-08-01' // Falla (fecha pasada) o devuelve lista vacía
mvn exec:java -Dexec.mainClass=es.udc.ws.app.client.ui.AppServiceClient "-Dexec.args=-findmatches '2023-08-01'T'00:00'"

6.Buscar Partido por identificador:

-findMatch 2:
mvn exec:java -Dexec.mainClass=es.udc.ws.app.client.ui.AppServiceClient "-Dexec.args=-findMatch '2'"

-findMatch 9: // Falla (partido no existe)
mvn exec:java -Dexec.mainClass=es.udc.ws.app.client.ui.AppServiceClient "-Dexec.args=-findMatch '9'"
